    <!-- JavaScript Logic Layer -->
    <script>
        // DOM Elements
        const vSlider = document.getElementById('valence');
        const eSlider = document.getElementById('energy');
        const aSlider = document.getElementById('acousticness');
        
        const vLabel = document.getElementById('valDisplay');
        const eLabel = document.getElementById('energyDisplay');
        const aLabel = document.getElementById('acousticDisplay');

        // Color Generation Mapping Logic
        function updatePalette() {
            const v = parseFloat(vSlider.value);
            const e = parseFloat(eSlider.value);
            const a = parseFloat(aSlider.value);

            // Update Dynamic Slider Labels
            vLabel.innerText = v < 0.35 ? "Melancholic" : v > 0.65 ? "Euphoric" : "Balanced";
            eLabel.innerText = e < 0.35 ? "Ambient" : e > 0.65 ? "Aggressive" : "Dynamic";
            aLabel.innerText = a < 0.35 ? "Electronic" : a > 0.65 ? "Organic" : "Hybrid";

            // Math to translate variables to cohesive HSL strings
            const baseHue = Math.floor((v * 180) + 220) % 360; 
            const saturation = Math.floor((e * 50) + 40);      
            const lightness = Math.floor((a * 30) + 35);       

            const colors = [
                `hsl(${baseHue}, ${saturation}%, ${lightness}%)`,                         
                `hsl(${(baseHue + 25) % 360}, ${saturation}%, ${lightness - 8}%)`,       
                `hsl(${(baseHue + 180) % 360}, ${Math.max(15, saturation - 25)}%, 82%)`, 
                `hsl(${(baseHue + 110) % 360}, ${Math.min(100, saturation + 15)}%, 18%)`  
            ];

            // Apply colors directly to CSS grid items
            for (let i = 1; i <= 4; i++) {
                const element = document.getElementById(`color${i}`);
                const textElement = document.getElementById(`hex${i}`);
                element.style.backgroundColor = colors[i - 1];
                textElement.innerText = colors[i - 1];
            }
        }

        // Attach listeners for immediate slider feedback
        vSlider.addEventListener('input', updatePalette);
        eSlider.addEventListener('input', updatePalette);
        aSlider.addEventListener('input', updatePalette);

        // Run once on load to populate the initial UI state
        updatePalette();
    </script>
